import java.util.Scanner;
class Main {
    public static void main(String[] args) {
        Scanner k = new Scanner(System.in);
        int row = k.nextInt();
        int cols = 2*row-1;
        int spaces = 0;
        // the first part
        int x = row;
        // the second part
        int y = cols - row;
        
        // the outer loop is for the number of rows
        for (int i = 0;i<row;i++){
            // print spaces
            for (int s = 0; s<spaces;s++){
                System.out.print(" ");
            }
            
            // print the first part of the input
            for (int a = 0; a<x;a++){
                // TO DO take a separate var that will increase
                System.out.print();
            }
            
            // print the second part of the input
            for (int b = 0; b<y;b++){
                // TO DO take a separate var that will decrease
                System.out.print();
            }
            spaces++;
            x--;
            y--;
            System.out.println("");
        }
    }
}
